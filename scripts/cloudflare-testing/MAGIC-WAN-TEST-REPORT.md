### 2. IPsec Configuration
```bash
# Create /etc/ipsec/ipsec.conf
config setup
  uniqueids=no

conn magic-wan-primary
  keyexchange=ikev2
  type=tunnel
  auto=start
  left=%any
  leftid=95.163.250.190
  leftsubnet=10.0.0.0/24
  right=162.159.75.92
  rightid=162.159.75.92
  rightsubnet=0.0.0.0/0
  ike=aes256-sha256-modp2048!
  esp=aes256-sha256!
  ikeport=500
  fragmentation=yes
  forceencaps=yes
```
